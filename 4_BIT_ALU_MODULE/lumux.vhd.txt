entity lumux is
    Port ( lum0,lum1,lum2,lum3,lum4,lum5,lum6,lum7 : in STD_LOGIC_VECTOR (4 downto 0);
           sel : in STD_LOGIC_VECTOR (2 downto 0);
           lum_out: out STD_LOGIC_VECTOR(4 downto 0));
end lumux;

architecture Behavioral of lumux is

begin
process(sel, lum0, lum1, lum2, lum3, lum4, lum5, lum6, lum7)
begin
case sel is
when "000"=>lum_out<=lum0;
when "001"=>lum_out<=lum1;
when "010"=>lum_out<=lum2;
when "011"=>lum_out<=lum3;
when "100"=>lum_out<=lum4;
when "101"=>lum_out<=lum5;
when "110"=>lum_out<=lum6;
when "111"=>lum_out<=lum7;
when others =>lum_out<="00000"; 
end case;
end process;
end Behavioral;
