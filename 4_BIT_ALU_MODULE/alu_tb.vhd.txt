library IEEE;
use IEEE.STD_LOGIC_1164.ALL;

entity alu_tb is
end alu_tb;

architecture Behavioral of alu_tb is

    component alu
        Port (
            A   : in  STD_LOGIC_VECTOR (3 downto 0);
            B   : in  STD_LOGIC_VECTOR (3 downto 0);
            SEL : in  STD_LOGIC_VECTOR (3 downto 0);
            Z   : out STD_LOGIC_VECTOR (4 downto 0)
        );
    end component;

    signal A   : STD_LOGIC_VECTOR (3 downto 0);
    signal B   : STD_LOGIC_VECTOR (3 downto 0);
    signal SEL : STD_LOGIC_VECTOR (3 downto 0);
    signal Z   : STD_LOGIC_VECTOR (4 downto 0);

begin

    uut: alu
        port map(
            A   => A,
            B   => B,
            SEL => SEL,
            Z   => Z
        );

    stim_proc: process
    begin

        -- Test Values
        A <= "0101";   -- 5
        B <= "0011";   -- 3

        -- Arithmetic Unit (SEL(3)=0)
        SEL <= "0000"; wait for 20 ns; -- A+B
        SEL <= "0001"; wait for 20 ns; -- A-B
        SEL <= "0010"; wait for 20 ns; -- A+1
        SEL <= "0011"; wait for 20 ns; -- B+1
        SEL <= "0100"; wait for 20 ns; -- A-1
        SEL <= "0101"; wait for 20 ns; -- B-1
        SEL <= "0110"; wait for 20 ns; -- Buffer A
        SEL <= "0111"; wait for 20 ns; -- Buffer B

        -- Logic Unit (SEL(3)=1)
        SEL <= "1000"; wait for 20 ns; -- AND
        SEL <= "1001"; wait for 20 ns; -- OR
        SEL <= "1010"; wait for 20 ns; -- NOT A
        SEL <= "1011"; wait for 20 ns; -- NOT B
        SEL <= "1100"; wait for 20 ns; -- NAND
        SEL <= "1101"; wait for 20 ns; -- NOR
        SEL <= "1110"; wait for 20 ns; -- XOR
        SEL <= "1111"; wait for 20 ns; -- XNOR

        -- Another Test Case
        A <= "1111";
        B <= "0001";

        SEL <= "0000"; wait for 20 ns; -- Overflow addition
        SEL <= "0001"; wait for 20 ns; -- Subtraction
        SEL <= "1000"; wait for 20 ns; -- AND
        SEL <= "1110"; wait for 20 ns; -- XOR

        wait;

    end process;

end Behavioral;
