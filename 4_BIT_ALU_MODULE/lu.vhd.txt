library IEEE;
use IEEE.STD_LOGIC_1164.ALL;

entity lu is
    Port ( Ain : in STD_LOGIC_VECTOR (3 downto 0);
           Bin : in STD_LOGIC_VECTOR (3 downto 0);
           S : in STD_LOGIC_VECTOR (2 downto 0);
           lu_out : out STD_LOGIC_VECTOR (4 downto 0));
end lu;

architecture Behavioral of lu is
component lumux is
 Port ( lum0,lum1,lum2,lum3,lum4,lum5,lum6,lum7 : in STD_LOGIC_VECTOR (4 downto 0);
           sel : in STD_LOGIC_VECTOR (2 downto 0);
           lum_out: out STD_LOGIC_VECTOR(4 downto 0));
           end component;
signal and_out,or_out,anot_out,bnot_out,nand_out,nor_out,xor_out,xnor_out:std_logic_vector(4 downto 0);

begin
and_out(4)<='0' ;
and_out(3 downto 0)<=(Ain and Bin);
or_out(4)<='0' ;
or_out(3 downto 0)<=(Ain or Bin);
anot_out(4)<='0';
anot_out(3 downto 0)<=not(Ain);
bnot_out(4)<='0';
bnot_out(3 downto 0)<=not(Bin);
nand_out(4)<='0';
nand_out(3 downto 0)<=(Ain nand Bin);
nor_out(4)<='0';
nor_out(3 downto 0)<=(Ain nor Bin);
xor_out(4)<='0';
xor_out(3 downto 0)<=(Ain xor Bin);
xnor_out(4)<='0';
xnor_out(3 downto 0)<=(Ain xnor Bin);
lum1: lumux port map(
        lum0=>and_out,
        lum1=>or_out,
        lum2=>anot_out,
        lum3=>bnot_out,
        lum4=>nand_out,
        lum5=>nor_out,
        lum6=>xor_out,
        lum7=>xnor_out,
        sel=>S,
        lum_out=>lu_out);

end Behavioral;
