use IEEE.STD_LOGIC_1164.ALL;

entity alu is
    Port ( A : in STD_LOGIC_VECTOR (3 downto 0);
           B : in STD_LOGIC_VECTOR (3 downto 0);
           SEL : in STD_LOGIC_VECTOR (3 downto 0);
           Z : out STD_LOGIC_VECTOR (4 downto 0));
end alu;

architecture Behavioral of alu is
component out_mux is
Port(
I0,I1:in std_logic_vector(4 downto 0);
sel:in std_logic;
f_out:out std_logic_vector(4 downto 0));
end component;
component au is
Port(
    Ain: in STD_LOGIC_VECTOR(3 downto 0);
    Bin: in STD_LOGIC_VECTOR(3 downto 0);
    S: in STD_LOGIC_VECTOR(2 downto 0);
    au_out: out STD_LOGIC_VECTOR(4 downto 0));
end component;
component lu is
Port ( Ain : in STD_LOGIC_VECTOR (3 downto 0);
           Bin : in STD_LOGIC_VECTOR (3 downto 0);
           S : in STD_LOGIC_VECTOR (2 downto 0);
           lu_out : out STD_LOGIC_VECTOR (4 downto 0));
           end component;
           signal au_out,lu_out:std_logic_vector(4 downto 0);
begin
out_mu1: out_mux port map(
        f_out=>Z,
        I0=>au_out,
        I1=>lu_out,
        sel=>SEL(3));
 au1: au port map(
        Ain=>A,
        Bin=>B,
        S=>SEL(2 downto 0),
        au_out=>au_out);
 lu1: lu port map(
        Ain=>A,
        Bin=>B,
        S=>SEL(2 downto 0),
        lu_out=>lu_out);
               

end Behavioral;
