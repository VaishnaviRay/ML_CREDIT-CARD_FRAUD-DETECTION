entity out_mux is
Port(
I0,I1:in std_logic_vector(4 downto 0);
sel:in std_logic;
f_out:out std_logic_vector(4 downto 0));

end out_mux;

architecture Behavioral of out_mux is

begin
process(sel, I0, I1)
begin
case sel is
when '0'=>f_out<=I0;
when '1'=>f_out<=I1;
when others=>f_out<="00000";
end case;
end process;
end Behavioral;
