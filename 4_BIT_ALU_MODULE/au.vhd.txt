library IEEE;
use IEEE.STD_LOGIC_1164.ALL;

-- Uncomment the following library declaration if using
-- arithmetic functions with Signed or Unsigned values
--use IEEE.NUMERIC_STD.ALL;

-- Uncomment the following library declaration if instantiating
-- any Xilinx leaf cells in this code.
--library UNISIM;
--use UNISIM.VComponents.all;

entity au is
Port(
    Ain: in STD_LOGIC_VECTOR(3 downto 0);
    Bin: in STD_LOGIC_VECTOR(3 downto 0);
    S: in STD_LOGIC_VECTOR(2 downto 0);
    au_out: out STD_LOGIC_VECTOR(4 downto 0));
end au;

architecture Behavioral of au is

           
component adder4 is
        Port ( A: in STD_LOGIC_VECTOR(3 downto 0);
        B: in STD_LOGIC_VECTOR(3 downto 0);
        Cin: in STD_LOGIC;
        Sum: out STD_LOGIC_VECTOR(3 downto 0);
        Cout: out STD_LOGIC);
       end component;
component sub4 is
        Port ( A : in STD_LOGIC_VECTOR (3 downto 0);
           B : in STD_LOGIC_VECTOR (3 downto 0);
           Bin : in STD_LOGIC;
           D : out STD_LOGIC_VECTOR (3 downto 0);
           Bout : out STD_LOGIC);
           end component;
component incrementor is 
            Port ( A : in STD_LOGIC_VECTOR (3 downto 0);
           Y : out STD_LOGIC_VECTOR (3 downto 0);
           Cout : out STD_LOGIC);
           end component;
component decrementor is
            Port (
        A : in  STD_LOGIC_VECTOR(3 downto 0);
        Y : out STD_LOGIC_VECTOR(3 downto 0)
    );
    end component;
    component aumux is
     Port ( aum0,aum1,aum2,aum3,aum4,aum5,aum6,aum7 : in STD_LOGIC_VECTOR (4 downto 0);
           sel : in STD_LOGIC_VECTOR (2 downto 0);
           aum_out:out STD_LOGIC_VECTOR(4 downto 0));
           end component;
    signal add_out,sub_out:std_logic_vector(4 downto 0);
    signal ainc_out,adec_out,binc_out,bdec_out,abuf_out,bbuf_out: std_logic_vector(4 downto 0);     
       
begin
add: adder4 port map(
                A=>Ain,
                B=>Bin,
                Cin=>'0',
                Cout=>add_out(4),
                Sum=>add_out(3 downto 0));
sub: sub4 port map(
                A=>Ain,
                B=>Bin,
                Bin=>'0',
                D=>sub_out(3 downto 0),
                Bout=>sub_out(4));
 inca: incrementor port map(
                A=>Ain,
                Y=>ainc_out(3 downto 0),
                Cout=>ainc_out(4));
 deca: decrementor port map(
                A=>Ain,
                Y=>adec_out(3 downto 0));
                adec_out(4)<='0';                            
 incb: incrementor port map(
                A=>Bin,
                Y=>binc_out(3 downto 0),
                Cout=>binc_out(4));
 decb: decrementor port map(
                A=>Bin,
                Y=>bdec_out(3 downto 0));
                bdec_out(4)<='0';
  bbuf_out(3 downto 0)<=Bin;
  bbuf_out(4)<='0';
  abuf_out(4)<='0';
  abuf_out(3 downto 0)<=Ain; 
  aumu: aumux port map(
                aum0=>add_out,
                aum1=>sub_out,
                aum2=>ainc_out,
                aum3=>binc_out,
                aum4=>adec_out,
                aum5=>bdec_out,
                aum6=>abuf_out,
                aum7=>bbuf_out,
                sel=>S,
                aum_out=>au_out);                                 

end Behavioral;
