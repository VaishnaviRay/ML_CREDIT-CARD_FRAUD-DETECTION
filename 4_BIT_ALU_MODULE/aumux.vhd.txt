entity aumux is
    Port ( aum0,aum1,aum2,aum3,aum4,aum5,aum6,aum7 : in STD_LOGIC_VECTOR (4 downto 0);
           sel : in STD_LOGIC_VECTOR (2 downto 0);
           aum_out:out STD_LOGIC_VECTOR(4 downto 0));
end aumux;

architecture Behavioral of aumux is
begin
process(sel, aum0, aum1, aum2, aum3, aum4, aum5, aum6, aum7)
begin

case sel is
when "000"=>aum_out<=aum0;
when "001"=>aum_out<=aum1;
when "010"=>aum_out<=aum2;
when "011"=>aum_out<=aum3;
when "100"=>aum_out<=aum4;
when "101"=>aum_out<=aum5;
when "110"=>aum_out<=aum6;
when "111"=>aum_out<=aum7;
when others =>aum_out<="00000"; 
end case;
end process;

end Behavioral;
